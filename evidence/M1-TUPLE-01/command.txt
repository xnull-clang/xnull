# 3 双路径回归（既有 9 个示例；元组无既有示例）
cd impl && go build -o /tmp/xnull-tuple ./cmd/xnull
cd ..
for f in examples/*.x; do
  n=$(basename "$f" .x)
  /tmp/xnull-tuple run --interp "$f" > evidence/M1-TUPLE-01/dualpath/$n.interp.out 2> evidence/M1-TUPLE-01/dualpath/$n.interp.err
  /tmp/xnull-tuple run "$f" > evidence/M1-TUPLE-01/dualpath/$n.compiled.out 2> evidence/M1-TUPLE-01/dualpath/$n.compiled.err
  cmp evidence/M1-TUPLE-01/dualpath/$n.interp.out evidence/M1-TUPLE-01/dualpath/$n.compiled.out
done
