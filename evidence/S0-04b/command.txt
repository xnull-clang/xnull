# S0-04b 采集命令

```bash
# 1 语法与清单
bash -n ci/run.sh
bash ci/run.sh --list                 # 期望 10 行

# 2 第 10 项正反向
bash ci/run.sh --only 10              # 期望 PASS 0

touch xx.xn
bash ci/run.sh --only 10              # 期望 FAIL 1（点名 ./xx.xn）
rm xx.xn
bash ci/run.sh --only 10              # 期望 PASS 0

printf '示例入口：examples/hello.xn\n' > ci/probe-ref.md
git add ci/probe-ref.md
bash ci/run.sh --only 10              # 期望 FAIL 1（点名活引用）
git rm --cached ci/probe-ref.md && rm ci/probe-ref.md
bash ci/run.sh --only 10              # 期望 PASS 0

# 3 单步与全量
bash ci/run.sh --only 01
bash ci/run.sh --only 02
bash ci/run.sh                        # 期望 PASS 4 / SKIP 6 / exit 0

# 4 入库
git add ci/lib.sh ci/run.sh ci/README.md ci/.out/.gitignore
```
