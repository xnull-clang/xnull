// 自举素材 #3（M5.1-c）：XNull 实现的表达式解析器（表驱动 Pratt，子集）。
//
// 与 Go 实现（internal/xfront/parser.go）同输入对照：对同一组表达式，
// 输出与 xfront.PrintExpr 同形态的 S 表达式（(int N)/(ident x)/(binary op L R)/(unary op X)）。
// 括号在两侧都透明（不进树）。设计约束：Parser 直接持源串游标扫描
// （带参构造与容器字段赋值尚未放开，显式绕开——不静默降级）。
//
// 只使用已落地语言面：类 + String 字段赋值 + List<Node> 字段（M5.1-b）
// + 方法递归 + while/if + String 成员。完成定义锚点：与 cmd/xnpt 输出逐字节一致。

class Node {
    Int kind = 0;      // 1=int 2=ident 3=binary 4=unary 5=error 6=paren
    String text = "";
    List<Node> kids = [];
}

static Node mkInt(String text) {
    Node t = new Node();
    t.kind = 1;
    t.text = text;
    return t;
}

static Node mkIdent(String text) {
    Node t = new Node();
    t.kind = 2;
    t.text = text;
    return t;
}

static Node mkBin(String op, Node l, Node r) {
    Node t = new Node();
    t.kind = 3;
    t.text = op;
    t.kids.push(l);
    t.kids.push(r);
    return t;
}

static Node mkUn(String op, Node x) {
    Node t = new Node();
    t.kind = 4;
    t.text = op;
    t.kids.push(x);
    return t;
}

// precOf：二元优先级表（数值越大越紧；0 = 非二元）。
// 与 Go 侧分层序一致：|| < && < ==/!= < 关系 < +- < */%。
static Int precOf(String op) {
    if (op == "||") { return 1; }
    if (op == "&&") { return 2; }
    if (op == "==") { return 3; }
    if (op == "!=") { return 3; }
    if (op == "<") { return 4; }
    if (op == ">") { return 4; }
    if (op == "<=") { return 4; }
    if (op == ">=") { return 4; }
    if (op == "+") { return 5; }
    if (op == "-") { return 5; }
    if (op == "*") { return 6; }
    if (op == "/") { return 6; }
    if (op == "%") { return 6; }
    return 0;
}

class Parser {
    String src = "";
    Int pos = 0;

    void skipSpace() {
        while (!self.eof()) {
            String c = self.cur(1);
            if (" \t\n\r".indexOf(c) < 0) {
                break;
            }
            self.pos = self.pos + 1;
        }
    }

    Bool eof() {
        return self.pos >= self.src.length();
    }

    String cur(Int n) {
        Int end = self.pos + n;
        if (end > self.src.length()) {
            end = self.src.length();
        }
        return self.src.substring(self.pos, end - self.pos);
    }

    void skip(Int n) {
        self.pos = self.pos + n;
    }

    // curKind：跳过空白后判定当前 token 类别（0=结束 1=int 2=ident 3=op -1=非法）。
    Int curKind() {
        self.skipSpace();
        if (self.eof()) {
            return 0;
        }
        String c = self.cur(1);
        if ("0123456789".indexOf(c) >= 0) {
            return 1;
        }
        if ("abcdefghijklmnopqrstuvwxyz".indexOf(c.toLower()) >= 0) {
            return 2;
        }
        if ("+-*/%<>()!|&=".indexOf(c) >= 0) {
            return 3;
        }
        return -1;
    }

    // curText：当前 token 原文（标识符/数字取连续段；两字符 op 优先）。
    String curText() {
        Int k = self.curKind();
        if (k == 1 || k == 2) {
            Int p = self.pos;
            while (p < self.src.length()) {
                String ch = self.src.substring(p, 1);
                Bool digit = "0123456789".indexOf(ch) >= 0;
                Bool alpha = "abcdefghijklmnopqrstuvwxyz".indexOf(ch.toLower()) >= 0;
                if (!digit && !alpha) {
                    break;
                }
                p = p + 1;
            }
            return self.src.substring(self.pos, p - self.pos);
        }
        if (k == 3) {
            String two = self.cur(2);
            if (two == "==" || two == "!=" || two == "<=" || two == ">=" || two == "&&" || two == "||") {
                return two;
            }
            return self.cur(1);
        }
        return "";
    }

    // parseBinary：Pratt 主循环（左结合；min 为最小可绑定优先级）。
    // 关系级（p==4）不可链式（对齐 Go sema SYN-102）：刚绑定过关系又遇关系
    // 即断，残留 token 由调用方以「未消费到尾」显式报错，不静默成错树。
    Node parseBinary(Int min) {
        Node lhs = self.parseUnary();
        Int lastP = 0;
        while (true) {
            if (self.curKind() != 3) {
                break;
            }
            String op = self.curText();
            Int p = precOf(op);
            if (p <= min) {
                break;
            }
            if (p == 4 && lastP == 4) {
                break;
            }
            self.skip(op.length());
            Node rhs = self.parseBinary(p);
            lhs = mkBin(op, lhs, rhs);
            lastP = p;
        }
        return lhs;
    }

    // parseUnary：一元 - !（绑定紧于全部二元，子集内足够）。
    Node parseUnary() {
        if (self.curKind() == 3) {
            String c = self.cur(1);
            if (c == "-" || c == "!") {
                self.skip(1);
                return mkUn(c, self.parseUnary());
            }
        }
        return self.parsePrimary();
    }

    Node parsePrimary() {
        Int k = self.curKind();
        if (k == 1 || k == 2) {
            String t = self.curText();
            self.skip(t.length());
            if (k == 1) {
                return mkInt(t);
            }
            return mkIdent(t);
        }
        if (k == 3 && self.cur(1) == "(") {
            self.skip(1);
            Node e = self.parseBinary(0);
            if (self.curKind() == 3 && self.cur(1) == ")") {
                self.skip(1);
                return mkParen(e);
            }
            return mkErr(); // 缺右括号：显式错误节点（不静默吞）
        }
        return mkErr();
    }
}

static Node mkParen(Node x) {
    Node t = new Node();
    t.kind = 6;
    t.kids.push(x);
    return t;
}

static Node mkErr() {
    Node t = new Node();
    t.kind = 5;
    return t;
}

// dump：S 表达式形态（与 xfront.PrintExpr 对齐）。
static String dump(Node n) {
    if (n.kind == 1) {
        return "(int " + n.text + ")";
    }
    if (n.kind == 2) {
        return "(ident " + n.text + ")";
    }
    if (n.kind == 3) {
        return "(binary " + n.text + " " + dump(n.kids.get(0)) + " " + dump(n.kids.get(1)) + ")";
    }
    if (n.kind == 4) {
        return "(unary " + n.text + " " + dump(n.kids.get(0)) + ")";
    }
    if (n.kind == 5) {
        return "(error)";
    }
    if (n.kind == 6) {
        return "(paren " + dump(n.kids.get(0)) + ")";
    }
    return "(?)";
}

static void show(String s) {
    Parser p = new Parser();
    p.src = s;
    Node e = p.parseBinary(0);
    p.skipSpace();
    if (!p.eof()) {
        // 未消费到尾：输入有解析器不认的构造——显式报错，不静默丢尾
        print("(error trailing)");
        return;
    }
    print(dump(e));
}

static Int main() {
    show("1 + 2 * 3");
    show("(1 + 2) * 3");
    show("10 - 4 - 3");
    show("100 / 10 / 2");
    show("a || b && c");
    show("1 + 2 < 4 && x != y");
    show("-x * 2");
    show("!(a == b) != c");
    show("2 * (3 + 4) > 10 && x != y");
    show("-(1 + 2) % 5");
    show("1*2+3*4-5");
    show("((1))");
    show("- - x");
    show("!!y");
    show("1 < 2 == 3 < 4");
    show("a % b * c");
    show("(1 + 2) * (3 - 4) / 5");
    show("1 + -2 * -3");
    show("-(a) + (!(b)) % 2");
    show("1==1!=0");
    show("a&&(b||c)&&d");
    return 0;
}
