class Box<T> {
    final T value
    Box(T v) { self.value = v }
    Box<R> map<R>(R(T) transform) {
        return new Box<R>(transform(self.value))
    }
    R get<R>(R(T) transform) {
        return transform(self.value)
    }
}
class Tools {
    static R apply<T, R>(T value, R(T) transform) {
        return transform(value)
    }
}
static Int main() {
    Box<Int> box = new Box<Int>(7)
    Box<String> text = box.map((Int value) -> str(value))
    print(text.value)
    Box<Int> number = text.map((String value) -> parseInt(value).value())
    print(number.value)
    Box<String> explicit = box.map<String>((Int value) -> "v" + str(value))
    print(explicit.value)
    print(box.get((Int value) -> value * 2))
    print(Tools.apply(3, (Int value) -> value + 1))
    return 0
}
